--[[
    ESP com papéis para MM2 (Roblox) · executor com Drawing
    - caixa + nome + distância em todos os jogadores (atravessa parede)
    - vermelho: quem está com a faca (assassino)
    - azul: quem está com a arma (xerife)
    - verde: inocente
    - INSERT liga/desliga

    O papel vem do próprio jogo: o MM2 põe a Tool "Knife" ou "Gun" na Backpack
    do jogador. Uma vez visto, o papel fica preso ao nick até a rodada virar
    (a rodada vira quando o seu personagem respawna).
]]

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UIS = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer
local Camera = workspace.CurrentCamera

local COLORS = {
    murderer = Color3.fromRGB(255, 45, 63),
    sheriff = Color3.fromRGB(45, 123, 255),
    innocent = Color3.fromRGB(70, 217, 110),
}
local LABELS = { murderer = "FACA", sheriff = "ARMA", innocent = "" }
local KNIFE = "knife"
local GUNS = { gun = true, guns = true, revolver = true, pistol = true }

local enabled = true
local roles = {}          -- [player] = "murderer" | "sheriff"
local drawings = {}       -- [player] = { box, name }
local conn

-- ---------------------------------------------------------------- papel
local function toolRole(container)
    if not container then return nil end
    for _, item in ipairs(container:GetChildren()) do
        local name = item.Name:lower()
        if name == KNIFE then return "murderer" end
        if GUNS[name] then return "sheriff" end
    end
    return nil
end

local function refreshRoles()
    for _, player in ipairs(Players:GetPlayers()) do
        local character = player.Character
        local role = toolRole(character) or toolRole(player:FindFirstChildOfClass("Backpack"))
        if role then
            roles[player] = role
        end
    end
end

-- ------------------------------------------------------------- desenho
local function make(player)
    local d = drawings[player]
    if d then return d end
    d = {}
    d.box = Drawing.new("Square")
    d.box.Thickness = 1
    d.box.Filled = false
    d.name = Drawing.new("Text")
    d.name.Size = 14
    d.name.Center = true
    d.name.Outline = true
    d.name.OutlineColor = Color3.new(0, 0, 0)
    drawings[player] = d
    return d
end

local function hide(player)
    local d = drawings[player]
    if not d then return end
    d.box.Visible = false
    d.name.Visible = false
end

local function remove(player)
    local d = drawings[player]
    if d then
        d.box:Remove()
        d.name:Remove()
        drawings[player] = nil
    end
    roles[player] = nil
end

local function cleanAll()
    for player in pairs(drawings) do remove(player) end
    for player in pairs(roles) do roles[player] = nil end
end

local function project(pos)
    local screen, onScreen = Camera:WorldToViewportPoint(pos)
    if screen.Z <= 0 then return nil end
    return Vector2.new(screen.X, screen.Y)
end

-- --------------------------------------------------------------- ciclo
local accumulator = 0
local function update(dt)
    accumulator += dt
    if accumulator >= 0.4 then          -- papéis mudam devagar; 2,5x por segundo basta
        accumulator = 0
        refreshRoles()
    end

    if not enabled then return end
    Camera = workspace.CurrentCamera
    if not Camera then return end

    for _, player in ipairs(Players:GetPlayers()) do
        if player == LocalPlayer then
            hide(player)
            continue
        end
        local character = player.Character
        local hrp = character and character:FindFirstChild("HumanoidRootPart")
        local head = character and character:FindFirstChild("Head")
        local humanoid = character and character:FindFirstChildOfClass("Humanoid")
        if not (hrp and head) or not humanoid or humanoid.Health <= 0 then
            hide(player)
            continue
        end

        local top = project(head.Position + Vector3.new(0, 0.6, 0))
        local bottom = project(hrp.Position - Vector3.new(0, 2.6, 0))
        if not top or not bottom then
            hide(player)
            continue
        end

        local height = math.abs(bottom.Y - top.Y)
        if height < 6 then
            hide(player)
            continue
        end

        local role = roles[player] or "innocent"
        local color = COLORS[role]
        local width = height * 0.45
        local d = make(player)

        d.box.Size = Vector2.new(width, height)
        d.box.Position = Vector2.new(top.X - width / 2, top.Y)
        d.box.Color = color
        d.box.Visible = true

        local distance = math.floor((hrp.Position - Camera.CFrame.Position).Magnitude)
        local tag = LABELS[role]
        d.name.Text = tag ~= ""
            and string.format("%s [%s] %dm", player.Name, tag, distance)
            or string.format("%s %dm", player.Name, distance)
        d.name.Color = color
        d.name.Position = Vector2.new(top.X, top.Y - 16)
        d.name.Visible = true
    end

    for player in pairs(drawings) do
        if not player.Parent then remove(player) end
    end
end

conn = RunService.RenderStepped:Connect(update)

-- Rodada nova = personagem novo para você. Os papéis antigos não valem mais.
local function watchLocalCharacter()
    LocalPlayer.CharacterAdded:Connect(function()
        roles = {}
    end)
end
watchLocalCharacter()

UIS.InputBegan:Connect(function(input, processed)
    if processed then return end
    if input.KeyCode == Enum.KeyCode.Insert then
        enabled = not enabled
        if not enabled then
            for player in pairs(drawings) do hide(player) end
        end
    end
end)

Players.PlayerRemoving:Connect(remove)

local function shutdown()
    if conn then conn:Disconnect() end
    cleanAll()
end

return shutdown
